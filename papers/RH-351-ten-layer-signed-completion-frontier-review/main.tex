\begin{abstract}
We audit RH-342--RH-350 and close the batch with a growing-depth
information-class theorem.  On the physical clock, RH-348 and RH-350 give
the exact selected lower-even direct coefficients
\[
 p_{k,j}=Y_{k,j}+\cP_{k,j}-S_{k,j},
 \qquad m_{k,j}=k-j,
 \qquad 2\le j\le J_k,
\]
where $J_k\to\infty$ and $J_k=o(k)$.  RH-350 proves uniform normalized laws
for the deterministic demand $S$ and parity packet $\cP$, an exact weighted
minimax law, and a conditional exponentially divergent subprefix theorem
under an unproved aggregate estimate for the actual signed remainder $Y$.

The new theorem fixes the proved $\cP/S$ arrays and considers only their
abstract signed coefficient-ledger completions.  The affine identity is
surjective: every prescribed residual array $r$ is obtained from
$Y=S-\cP+r$.  In particular, $Y^{\rm close}=S-\cP$ gives zero selected
residual, while $Y^{\rm far}=0$ gives the RH-350 minimax residual.  Their
normalized budgets obey the exact exchange
\[
 \cY_k(Y^{\rm close})=\cL_k(Y^{\rm far}),
 \qquad
 \cY_k(Y^{\rm far})=\cL_k(Y^{\rm close})=0,
\]
and $\cL_k(Y^{\rm far})$ has a positive limiting lower bound.  Thus the
current deterministic/scalar information does not determine the actual
selected physical residual.  The two completions are not noisy operators,
Markov kernels, raw trace partitions, or determinant realizations.

The actual moving-order $Y$ theorem, head transport, critical and
first-lower compensation, odd and upper-alias control, and the full
$E_{\rm off}$ aggregate remain open.  RH-241's moving noisy all-order
envelope remains distinct from the later deterministic anchor and envelope.
RH-288 and Gates A--E are inactive.  No Riemann-hypothesis conclusion
follows.
\end{abstract}

\section{The exact selected coefficient and RH-350 input}

For each coordinate in \eqref{eq:window}, RH-348 gives the direct coefficient
identity
\begin{equation}\label{eq:direct}
 \boxed{
 p_{k,j}=Y_{k,j}+\cP_{k,j}-S_{k,j},}
 \qquad
 Y_{k,j}=\cT_{k,m_{k,j}}^{\rm rest}
 -d_{\sigma,k,2m_{k,j}}.
\end{equation}
The term $Y$ is signed and actual.  It combines the orbit-free physical rest
with the defect between the noisy head and the counterloop.  It is not a
free parameter of the physical problem.

Define the selected direct and remainder budgets
\begin{align}
 \cL_k(Y)&=\frac1{x^{k-2}}
 \sum_{j=2}^{J_k}
 \frac{|Y_{k,j}+\cP_{k,j}-S_{k,j}|}{2H_{m_{k,j}}},
 \label{eq:L}\\
 \cY_k(Y)&=\frac1{x^{k-2}}
 \sum_{j=2}^{J_k}
 \frac{|Y_{k,j}|}{2H_{m_{k,j}}}.
 \label{eq:Yagg}
\end{align}
Since $H_m=mR^{-2m}$, the unnormalized sum in \eqref{eq:L} is exactly the
selected lower-even contribution with prefix weight $R^{2m}/(2m)$.

Put
\begin{equation}\label{eq:F}
 F_N(a)=\sum_{r=0}^{N}x^{-r}|a\lambda^{-r}-1|.
\end{equation}
RH-350 proves that $a=1$ is its unique minimizer and
\begin{align}
 A_N:=\inf_{a>0}F_N(a)
 &=\frac{1-x^{-N}}{x-1}
 -\frac{1-(x\lambda)^{-N}}{x\lambda-1},
 \label{eq:AN}\\
 A_N&\nearrow A_\infty
 =\frac1{x-1}-\frac1{x\lambda-1}>0.
 \label{eq:Ainf}
\end{align}
Its conditional physical theorem says
\begin{equation}\label{eq:RH350}

 \quad\Longrightarrow\quad

and hence $\liminf\cL_k(Y^{\rm actual})\ge A_\infty/C_M$.
The antecedent in \eqref{eq:RH350} is not proved in the repository.

\section{Growing-depth signed-completion underdetermination}

\begin{definition}[selected coefficient-ledger information class]
Fix the source-locked arrays $S_{k,j}$ and $\cP_{k,j}$ satisfying
\eqref{eq:S-uniform}--\eqref{eq:P-uniform}.  The selected coefficient-ledger
information class consists of formal signed arrays $Y_{k,j}$ on
\eqref{eq:window}, with $p_{k,j}$ defined by \eqref{eq:direct}.  No Markov
kernel, transfer operator, raw trace partition, determinant, or cross-order
physical realizability condition is added.
\end{definition}

\begin{proposition}[affine completion surjectivity]
For every formal residual array $r_{k,j}$ on the selected window, the unique
coefficient-ledger completion
\begin{equation}\label{eq:affine}
 Y^{[r]}_{k,j}=S_{k,j}-\cP_{k,j}+r_{k,j}
\end{equation}
satisfies $p^{[r]}_{k,j}=r_{k,j}$ exactly.
\end{proposition}
\begin{proof}
Substitution of \eqref{eq:affine} into \eqref{eq:direct} gives
$p^{[r]}=(S-\cP+r)+\cP-S=r$ coordinatewise.  Conversely,
\eqref{eq:direct} can be solved uniquely for $Y$ once $p=r$ is prescribed.
\end{proof}

The proposition is deliberately algebraic.  It identifies precisely what
the proved $S/\cP$ information cannot constrain before a theorem for the
actual $Y$ is supplied.

\begin{theorem}[opposite growing-depth coefficient-ledger completions]
Within the selected coefficient-ledger information class, define
\begin{align}
 Y^{\rm close}_{k,j}&=S_{k,j}-\cP_{k,j},
 &p^{\rm close}_{k,j}&=0,
 \label{eq:close}\\
 Y^{\rm far}_{k,j}&=0,
 &p^{\rm far}_{k,j}&=\cP_{k,j}-S_{k,j}.
 \label{eq:far}
\end{align}
Then the normalized budgets obey the exact exchange
\begin{equation}\label{eq:exchange}
 \boxed{
 \cY_k(Y^{\rm close})=\cL_k(Y^{\rm far}),
 \qquad
 \cY_k(Y^{\rm far})=\cL_k(Y^{\rm close})=0.}
\end{equation}
Moreover, with $N_k=J_k-2$,
\begin{equation}\label{eq:far-law}
 \boxed{
 \cL_k(Y^{\rm far})
 =\frac{F_{N_k}(a_k)}{C_M}+o(1),}
\end{equation}
and
\begin{equation}\label{eq:far-liminf}
 \boxed{
 \liminf_{k\to\infty}\cL_k(Y^{\rm far})
 \ge\frac1{C_M}
 \left(\frac1{x-1}-\frac1{x\lambda-1}\right)>0.}
\end{equation}
Consequently the unnormalized selected sum in the far ledger diverges at
least on the exponential scale $x^{k-2}$, whereas the close ledger vanishes
identically on the selected window.
\end{theorem}

\begin{proof}
Equations \eqref{eq:close}--\eqref{eq:far} are the cases $r=0$ and
$r=\cP-S$ of affine surjectivity.  The identities
$|Y^{\rm close}|=|S-\cP|=|p^{\rm far}|$ and
$Y^{\rm far}=p^{\rm close}=0$ hold coordinatewise.  Multiplication by the
same positive weights in \eqref{eq:L}--\eqref{eq:Yagg} and summation prove
\eqref{eq:exchange}.

For the far ledger, \eqref{eq:S-uniform}--\eqref{eq:P-uniform} give,
uniformly in $j$,
\[
 \frac{|\cP_{k,j}-S_{k,j}|}
 {2H_{m_{k,j}}x^{m_{k,j}}}
 =\frac{|a_k\lambda^{2-j}-1|}{C_M}+o(1).
\]
Since $x^{m_{k,j}}/x^{k-2}=x^{2-j}$ and
$\sum_{j=2}^{\infty}x^{2-j}<\infty$, summing the uniform error gives
$o(1)$.  Reindexing by $r=j-2$ proves \eqref{eq:far-law} directly for the
formal ledger.  Equations \eqref{eq:AN}--\eqref{eq:Ainf} then give
\eqref{eq:far-liminf}.  Since $x>1$, multiplication by $x^{k-2}$ yields the
unnormalized divergence.  The close statement is exact from
\eqref{eq:close}.
\end{proof}

\begin{corollary}[the missing $Y$ theorem is decisive]
The proved $S/\cP$ arrays, their growing-window uniform laws, and their exact
minimax formulas determine neither vanishing nor nonvanishing of the actual
selected lower-even residual.  Any physical verdict requires an additional
moving-order theorem that restricts the actual $Y_{k,j}$.
\end{corollary}

\begin{proof}
The two formal ledgers satisfy the same fixed $S/\cP$ premises and have
opposite selected outcomes.  Therefore neither outcome is a logical
consequence of those premises alone.  The close ledger does not contradict
RH-350: by \eqref{eq:exchange}, its $Y$ budget equals the positive far
residual budget and hence fails the small-$Y$ hypothesis.
\end{proof}

\begin{remark}[strict interpretation]
The theorem constructs no physical noisy object.  In the physical problem,
$Y=\cT^{\rm rest}-d$ is fixed by one noisy dynamics and must satisfy any
cross-order identities that such a realization imposes.  Those constraints
are exactly what the current repository does not estimate.  The theorem is
a coefficient-information insufficiency result, not a physical existence or
nonexistence theorem.
\end{remark}
