\begin{abstract}
We audit RH-332--RH-340 on one physical natural clock, one Hardy
normalization, and the common determinant cut $u=4k$.  The deterministic
frontier left open by RH-241 has since split cleanly: RH-263 proves the
all-order deterministic numerator anchor, RH-267 proves a certified
all-order geometric envelope, and RH-268 proves its sharp coefficient
radius.  These results do not supply the moving noisy all-order coefficient
bridge or Gate A.

For the corrected physical observation type, the source chain is
\[
 q_n=\cB_n+\cS_n+\cR_n+\cP_n-\cA_{k,n},
 \qquad p_n=q_n-d_n,
 \qquad |P_u-E_u|\le D_u.
\]
At $u=4k$, both analytic tails vanish, while direct-prefix closure still
requires the same-clock conjunction
$D_{4k}\to0$, $E_{{\rm off},4k}\to0$, and
$q_{\sigma,k,2k}=o(H_k)$.  The physical boundary-orbit atoms at orders
$2k$ and $2k-2$ force two exponentially precise signed compensation
equations.  Separate absolute splitting of orbit, diffuse, and head pieces
has a divergent two-atom submajorant.

The new synthesis theorem is strictly about the available information
class.  Its unconstrained signed combined complements admit an abstract
cancelling ledger completion and an abstract noncancelling ledger
completion.  Thus the current exact identities and asymptotic atom sizes
determine neither aggregate prefix closure nor nonclosure.  These ledgers
are not two physical noisy operators, and no realizability claim is made.
The physical prefix, off-alias background, head budget, determinant gluing,
and Gates A--E remain open.  No Riemann-hypothesis conclusion follows.
\end{abstract}

\section{Exact synchronized prefix reduction}

For every $n\ge2$, the corrected all-order constituent is
\begin{equation}\label{eq:five-slot}
 q_{\sigma,k,n}
 =\cB_{\sigma,k,n}+\cS_{\sigma,k,n}+\cR_{\sigma,k,n}
  +\cP_{\sigma,n}-\cA_{k,n}.
\end{equation}
RH-339 states this all-order proposition explicitly; RH-334 supplies the
underlying all-order raw partition and the first-alias typed observation.
Let
\begin{equation}\label{eq:direct-identity}
 p_{\sigma,k,n}:=\tau_{\sigma,n}-a_n
 =q_{\sigma,k,n}-d_{\sigma,k,n},
 \qquad d_{\sigma,k,n}=h_{\sigma,n}-s_{k,n}.
\end{equation}
At a common finite cut $u$, define
\begin{align}\label{eq:budgets}
 P_u(R)&=\sum_{2\le n<u}\frac{|p_{\sigma,k,n}|R^n}{n},&
 E_u(R)&=\sum_{2\le n<u}\frac{|q_{\sigma,k,n}|R^n}{n},&
 D_u(R)&=\sum_{2\le n<u}\frac{|d_{\sigma,k,n}|R^n}{n}.
\end{align}

\begin{proposition}[same-clock prefix synchronization]
For every common finite cut and radius,
\begin{equation}\label{eq:sync}
 |P_u(R)-E_u(R)|\le D_u(R).
\end{equation}
Consequently $D_u(R)\to0$ implies
$P_u(R)\to0$ if and only if $E_u(R)\to0$.
\end{proposition}
\begin{proof}
Equation \eqref{eq:direct-identity} and the reverse triangle inequality give
$\bigl||p_n|-|q_n|\bigr|\le|d_n|$ at each order.  Multiply by $R^n/n$ and
sum.  This is the exact RH-340 argument.
\end{proof}

At $u=4k$, RH-340 re-applies the RH-282 mass estimate and the deterministic
envelope \eqref{eq:target-envelope} to prove that both analytic tails vanish.
The one-alias extraction is
\begin{equation}\label{eq:critical-extraction}
 E_{4k}(R)=E_{{\rm off},4k}(R)
 +\frac{|q_{\sigma,k,2k}|}{2H_k}.
\end{equation}
Hence the remaining sufficient input for the RH-288 prefix leaf is exactly
\begin{equation}\label{eq:three-budget}
 D_{4k}(R)\to0,
 \qquad E_{{\rm off},4k}(R)\to0,
 \qquad q_{\sigma,k,2k}=o(H_k).
\end{equation}
No term in \eqref{eq:three-budget} is proved by the current batch.  Tail
closure alone therefore does not activate determinant gluing.

\section{Two mandatory signed compensations}

Let $m=k-1$.  RH-338 and RH-339 prove physical signed orbit constituents
\begin{align}\label{eq:atoms}
 \cR_{k,\mathrm{orb}}&=-D_k^{\rm orb},&
 \frac{D_k^{\rm orb}}{H_k}
 &=\frac{2}{C_M}(\beta R)^{2k}\{1+o(1)\}\to+\infty,\\
 \cR_{k,-,\mathrm{orb}}&=-D_m^{\rm orb},&
 \frac{D_m^{\rm orb}}{H_m}
 &=\frac{2}{C_M}(\beta R)^{2m}\{1+o(1)\}\to+\infty.
\end{align}
Here $H_m=mR^{-2m}$ and $\beta R>1$ has an exact repository certificate.
Define the signed complements inside the Hardy full-trace coefficients by
\begin{equation}\label{eq:complements}
 C_k^0=q_{\sigma,k,2k}+D_k^{\rm orb},
 \qquad
 C_k^-=q_{\sigma,k,2m}+D_m^{\rm orb}.
\end{equation}
Using \eqref{eq:direct-identity}, the two direct coefficients are exactly
\begin{align}\label{eq:two-residuals}
 p_{\sigma,k,2k}&=-D_k^{\rm orb}+C_k^0-d_{\sigma,k,2k},\\
 p_{\sigma,k,2m}&=-D_m^{\rm orb}+C_k^- -d_{\sigma,k,2m}.
\end{align}

\begin{theorem}[necessary adjacent-order compensation]
If $P_{4k}(R)\to0$, then
\begin{align}\label{eq:compensations}
 C_k^0-d_{\sigma,k,2k}&=D_k^{\rm orb}+o(H_k),\\
 C_k^- -d_{\sigma,k,2k-2}&=D_{k-1}^{\rm orb}+o(H_{k-1}).
\end{align}
The relative precisions are respectively
$o((\beta R)^{-2k})$ and $o((\beta R)^{-2(k-1)})$.  If also
$D_{4k}(R)\to0$, then the two displayed head terms are negligible on their
target scales.
\end{theorem}
\begin{proof}
Both orders occur in the nonnegative prefix, and
\[
 P_{4k}(R)\ge
 \frac{|p_{\sigma,k,2k}|}{2H_k}
 +\frac{|p_{\sigma,k,2m}|}{2H_m}.
\]
Prefix closure makes the two numerators $o(H_k)$ and $o(H_m)$.
Substitute \eqref{eq:two-residuals}.  Division by the two asymptotics in
\eqref{eq:atoms} gives the relative precisions.  The same
nonnegative-summand argument applied to $D_{4k}$ makes the two head terms
negligible.
\end{proof}

Taking separate absolute values before the signed sums necessarily contains
\begin{equation}\label{eq:absolute-stop}
 \frac{D_k^{\rm orb}}{2H_k}+\frac{D_m^{\rm orb}}{2H_m}
 =\frac1{C_M}\left((\beta R)^{2k}+(\beta R)^{2k-2}\right)
 \{1+o(1)\}\longrightarrow+\infty.
\end{equation}
Thus the cancellation-blind orbit/diffuse/head route is stopped.  Equation
\eqref{eq:absolute-stop} is not a lower bound for the fully signed prefix.

\section{Abstract signed-completion underdetermination}

Put
\begin{equation}\label{eq:X}
 X_k^0=C_k^0-d_{\sigma,k,2k},
 \qquad X_k^-=C_k^- -d_{\sigma,k,2k-2}.
\end{equation}
The current source information fixes the two atoms and the exact identities
\begin{equation}\label{eq:abstract-ledger}
 p_{\sigma,k,2k}=-D_k^{\rm orb}+X_k^0,
 \qquad
 p_{\sigma,k,2k-2}=-D_{k-1}^{\rm orb}+X_k^-,
\end{equation}
but supplies no moving-order estimate for either signed combined complement.
Define the \emph{source information class} to consist of abstract signed
coefficient ledgers satisfying the proved identities, scales, and data-type
labels, with no additional physical realizability assumption.

\begin{theorem}[opposite abstract completions]
Within the source information class, neither $P_{4k}(R)\to0$ nor
$P_{4k}(R)\not\to0$ is logically determined.  More sharply, there is an
abstract completion with zero direct prefix and another whose two mandatory
orders alone diverge in the weighted prefix.  Both may be chosen with
$d_{\sigma,k,n}=0$ at every order.
\end{theorem}
\begin{proof}
For the cancelling completion, set
\[
 X_k^0=D_k^{\rm orb},
 \qquad X_k^-=D_{k-1}^{\rm orb},
\]
and set every other direct prefix coefficient to zero.  Equations
\eqref{eq:abstract-ledger} give $P_{4k}=0$.

For the noncancelling completion, set $X_k^0=X_k^-=0$ and again set every
other direct prefix coefficient to zero.  Then
\[
 P_{4k}(R)\ge
 \frac{D_k^{\rm orb}}{2H_k}
 +\frac{D_{k-1}^{\rm orb}}{2H_{k-1}}\longrightarrow+\infty
\]
by \eqref{eq:absolute-stop}.  Taking $d=0$ realizes both choices as abstract
full-trace complement ledgers.  The current source class imposes no theorem
that excludes either signed choice, so neither aggregate alternative is a
logical consequence of the available information.
\end{proof}

\begin{remark}[strict interpretation]
The proof constructs no Markov kernel, transfer operator, noisy quadratic
map, or determinant.  It does not assert that either completion is
physically realizable.  Its content is exactly the insufficiency of the
currently proved signed information.  A physical verdict requires a new
moving-order theorem for the actual combined complements and head defect.
\end{remark}
