\begin{abstract}
Let $3=p_1<p_2<\cdots$ be the odd primes, put $x=p_y$ and
$L=\log x$, and define the strict tails
\[
 P_r(y)=\sum_{p>x}(p^2-1)^{-r}\qquad(r\ge1).
\]
Using the explicit Vinogradov--Korobov estimate of Johnston and Yang, we
prove a quantitative comparison of $P_r$ with three successive kernels.
With
\[
 V=L^{3/5}(\log L)^{-1/5},\qquad
 \eps_x=0.027L^{1.801}e^{-0.1853V},
\]
the exact Stieltjes kernel $J_r$, power kernel $I_{2r}$, and leading term
$K_r=x^{1-2r}/((2r-1)L)$ satisfy
\[
 \left|\log\frac{P_r}{J_r}\right|\le14r\eps_x,
 \quad
 0\le\log\frac{J_r}{I_{2r}}\le\frac{r}{x^2-1},
 \quad
 \left|\log\frac{I_{2r}}{K_r}\right|
 \le\frac{1}{(2r-1)L},
\]
whenever $L\ge512$ and $7r\eps_x\le1/2$.
For a partition $\lambda=1^{k_1}2^{k_2}\cdots$, set
$d=\sum rk_r$, $H=\sum k_r/(2r-1)$, and
$H_2=\sum k_r/(2r-1)^2$.  The corresponding product has the refined
uniform expansion
\[
 \log\frac{\cP_\lambda}{\cM_\lambda}
 =-\frac{H}{L}
 +O\!\left(d\eps_x+\frac{d}{x^2-1}+\frac{H_2}{L^2}\right),
\]
with explicit constants $14,1,2$.  Thus
$d\eps_x+d/x^2\to0$ is a robust source-and-power condition, while the
leading equivalent holds precisely when $H/L\to0$.  In particular
$\log d=o(V)$ and $H=o(L)$ suffice.  For a single order the range
$\log R=o(V)$ is uniform, and one may sharpen it to
$R\le e^{(0.1853-\delta)V}$ for fixed $0<\delta<0.1853$.
The family $\lambda=1^{\lfloor cL\rfloor}$ has leading ratio tending to
$e^{-c}$, proving the $H/L$ obstruction is real.  An exact 96-row artifact
reproduces the algebraic interfaces but is not an analytic proof.
\end{abstract}

\noindent\textbf{Keywords:} explicit prime number theorem;
Vinogradov--Korobov estimate; strict Stieltjes boundary; prime-square tail;
growing integer partitions; uniform asymptotics.

\section{Scope and notation}

The fixed-order asymptotic
\[
 P_r(y)\sim\frac{x^{1-2r}}{(2r-1)\log x}
\]
follows from the prime number theorem when $r$ is fixed.  That statement
does not by itself permit $r$, the degree of a partition, or the number of
factors to grow with $y$.  The purpose of this paper is to pay for those
three kinds of growth explicitly.  The new input is the global explicit
Vinogradov--Korobov estimate of Johnston and Yang
\citep[Theorem~1.4, equation~(1.8)]{JohnstonYang2023}; the prior fixed-order
prime-tail calculation in RH-384 is used only as a finite regression target
for the accompanying artifact \citep{RH384}.

Throughout,
\[
 3=p_1<p_2<\cdots,\qquad x=p_y,\qquad L=\log x,
\]
and $r$ is a positive integer.  Write
\[
 \vartheta(t)=\sum_{p\le t}\log p,
 \qquad E(t)=\vartheta(t)-t.
\]
The strict prime tail and its three comparison kernels are
\begin{align}
 P_r(y)&=\sum_{p>x}\frac{1}{(p^2-1)^r},
 \label{eq:Pr}\\
 h_r(t)&=\frac{(t^2-1)^{-r}}{\log t},
 &J_r(x)&=\int_x^\infty h_r(t)\,dt,
 \label{eq:Jr}\\
 I_{2r}(x)&=\int_x^\infty\frac{t^{-2r}}{\log t}\,dt,
 &K_r(x)&=\frac{x^{1-2r}}{(2r-1)L}.
 \label{eq:IK}
\end{align}
The endpoint in \eqref{eq:Pr} is always strict.  In particular,
\begin{equation}
 P_r(y)=\frac{1}{(p_{y+1}^2-1)^r}+P_r(y+1).
 \label{eq:successor}
\end{equation}
This exact successor identity distinguishes $p>x$ from $p\ge x$ even
though their leading asymptotics agree.

Set
\begin{equation}
 V(L)=L^{3/5}(\log L)^{-1/5},\qquad
 \eps_x=\frac{27}{1000}L^{1801/1000}
 \exp\!\left(-\frac{1853}{10000}V(L)\right).
 \label{eq:epsilon}
\end{equation}
Johnston and Yang prove that, for every $t\ge23$,
\begin{equation}
 |E(t)|\le t\eps_t.
 \label{eq:JY}
\end{equation}
Their Corollary~1.2, equation~(1.5), and the $X=\log2$ row of their
Table~1 are retained in the external source lock as provenance locators,
but the derived square-root-exponential fallback is not used or asserted
as a result here.  Only \eqref{eq:JY} enters the proof.

\begin{proposition}[Monotone source envelope]
\label{prop:eta-monotone}
The function of $L$ on the right of \eqref{eq:epsilon} is strictly
decreasing for $L\ge512$.
\end{proposition}

\begin{proof}
Logarithmic differentiation gives
\[
 \frac{d}{dL}\log\eps
 =\frac1L\left[
 \frac{1801}{1000}
 -\frac{1853}{10000}V(L)
 \left(\frac35-\frac{1}{5\log L}\right)
 \right].
\]
For $L\ge512$, one has $\log L>6$, $\sqrt L>22$, and
$V(L)\ge\sqrt L$.  The last inequality is equivalent to
$\sqrt L\ge\log L$, which holds on this range.  Hence the factor in
parentheses is at least $17/30$, and
\[
 \frac{1853}{10000}\,22\,\frac{17}{30}
 -\frac{1801}{1000}>0.
\]
The displayed derivative is therefore negative.
\end{proof}

\section{The strict Stieltjes transfer}

The endpoint and the boundary term must be kept exact before applying the
source estimate.

\begin{lemma}[Strict Stieltjes identity]
\label{lem:stieltjes}
For $x=p_y$ and $r\ge1$,
\begin{equation}
 P_r(y)=\int_{(x,\infty)}h_r(t)\,d\vartheta(t)
 =-\vartheta(x)h_r(x)-\int_x^\infty\vartheta(t)h_r'(t)\,dt.
 \label{eq:stieltjes}
\end{equation}
Consequently,
\begin{equation}
 P_r(y)-J_r(x)
 =-E(x)h_r(x)-\int_x^\infty E(t)h_r'(t)\,dt.
 \label{eq:error-identity}
\end{equation}
\end{lemma}

\begin{proof}
At a prime $p$, the jump of $\vartheta$ is $\log p$, so the first
integral equals \eqref{eq:Pr}.  Stieltjes integration by parts on
$(x,R]$ gives the second expression with the lower boundary
$-\vartheta(x)h_r(x)$.  The upper boundary tends to zero; for example
$\vartheta(R)\le R\log R$ and $h_r(R)=O(R^{-2r}/\log R)$.
Substitute $\vartheta(t)=t+E(t)$ and use
\[
 -xh_r(x)-\int_x^\infty t h_r'(t)\,dt
 =\int_x^\infty h_r(t)\,dt=J_r(x)
\]
to obtain \eqref{eq:error-identity}.
\end{proof}

\begin{lemma}[Decreasing hazard]
\label{lem:hazard}
The logarithmic hazard
\begin{equation}
 q_r(t)=-\frac{h_r'(t)}{h_r(t)}
 =\frac{2rt}{t^2-1}+\frac{1}{t\log t}
 \label{eq:hazard}
\end{equation}
is positive and strictly decreasing for $t>1$.  Moreover,
\begin{equation}
 J_r(x)\ge\frac{h_r(x)}{q_r(x)},\qquad
 xq_r(x)\le3r\quad(x\ge23).
 \label{eq:hazard-bounds}
\end{equation}
\end{lemma}

\begin{proof}
Both summands in \eqref{eq:hazard} are positive and strictly decreasing:
their derivatives have numerators $-2r(t^2+1)$ and
$-(\log t+1)$, respectively, after multiplication by positive
denominators.  Hence
\[
 h_r(x)=\int_x^\infty q_r(t)h_r(t)\,dt
 \le q_r(x)J_r(x).
\]
Finally
\[
 xq_r(x)=\frac{2rx^2}{x^2-1}+\frac1{\log x}
 \le r\left(2+\frac{2}{528}+\frac13\right)<3r
\]
for $x\ge23$.
\end{proof}

\begin{proposition}[Explicit source-to-kernel transfer]
\label{prop:source-transfer}
Assume $L\ge512$.  Then
\begin{equation}
 \left|\frac{P_r(y)}{J_r(x)}-1\right|\le7r\eps_x.
 \label{eq:relative-source}
\end{equation}
If in addition $7r\eps_x\le1/2$, then
\begin{equation}
 \boxed{\left|\log\frac{P_r(y)}{J_r(x)}\right|
 \le14r\eps_x.}
 \label{eq:log-source}
\end{equation}
\end{proposition}

\begin{proof}
Since $h_r'<0$, Proposition~\ref{prop:eta-monotone},
\eqref{eq:JY}, and \eqref{eq:error-identity} imply
\begin{align*}
 |P_r-J_r|
 &\le\eps_x\left(xh_r(x)+
       \int_x^\infty t(-h_r'(t))\,dt\right)\\
 &=\eps_x\bigl(2xh_r(x)+J_r(x)\bigr).
\end{align*}
Lemma~\ref{lem:hazard} gives $xh_r(x)/J_r(x)\le3r$, so the relative
error is at most $(6r+1)\eps_x\le7r\eps_x$.  If this is at most $1/2$,
then $|\log(1+u)|\le2|u|$ for $|u|\le1/2$, proving
\eqref{eq:log-source}.
\end{proof}

\begin{remark}[Why the boundary is not optional]
The two copies of $xh_r(x)$ in the proof have different origins: one is
the strict Stieltjes boundary and one comes from
$\int t(-h_r')=xh_r+J_r$.  Deleting the boundary changes the transfer
algebra.  Conversely, the leading asymptotic alone cannot detect an
inclusive-endpoint mutation, which is why \eqref{eq:successor} is also
retained.
\end{remark}

\section{Two kernel comparisons}

\begin{lemma}[Exact kernel to power kernel]
\label{lem:J-I}
For every $x>1$ and $r\ge1$,
\begin{equation}
 0\le\log\frac{J_r(x)}{I_{2r}(x)}
 \le-r\log(1-x^{-2})
 \le\frac{r}{x^2-1}.
 \label{eq:J-I}
\end{equation}
No smallness assumption on $r/x^2$ is required.
\end{lemma}

\begin{proof}
The quotient $J_r/I_{2r}$ is the weighted average, with positive weight
$t^{-2r}/\log t$, of $(1-t^{-2})^{-r}$.  That factor decreases from
$(1-x^{-2})^{-r}$ to $1$.  This gives the first two inequalities.
The last follows from $-\log(1-u)\le u/(1-u)$ with $u=x^{-2}$.
\end{proof}

The optional coarser estimate $\log(J_r/I_{2r})\le4r/x^2$ used in an
earlier finite interface follows immediately on the present range (and in
particular under the formerly declared condition $r/x^2\le3/8$).  It is
not used in the canonical ledger.

\begin{lemma}[Power kernel to leading kernel]
\label{lem:I-K}
Put
\[
 a_r=\frac{1}{(2r-1)L},\qquad
 G(a)=\int_0^\infty\frac{e^{-v}}{1+av}\,dv.
\]
Then
\begin{equation}
 \frac{I_{2r}(x)}{K_r(x)}=G(a_r),\qquad
 \frac1{1+a}\le G(a)\le1,
 \label{eq:G-bounds}
\end{equation}
and therefore
\begin{equation}
 \left|\log\frac{I_{2r}(x)}{K_r(x)}\right|
 \le a_r=\frac{1}{(2r-1)L}.
 \label{eq:I-K}
\end{equation}
For $0\le a\le1/4$ one also has
\begin{equation}
 0\le\log G(a)+a\le2a^2.
 \label{eq:G-refined}
\end{equation}
\end{lemma}

\begin{proof}
Set $t=xe^{v/(2r-1)}$ in $I_{2r}$.  This gives the identity in
\eqref{eq:G-bounds}.  If $V_0$ is exponentially distributed with mean
one, then $G(a)=\mathbb E(1+aV_0)^{-1}$.  Jensen's inequality and the
trivial upper bound give $1/(1+a)\le G(a)\le1$, hence
$|\log G(a)|\le\log(1+a)\le a$.

For the refinement, $1/(1+a)\ge e^{-a}$ implies
$\log G(a)+a\ge0$.  The pointwise inequality
$(1+z)^{-1}\le1-z+z^2$ and the exponential moments
$\mathbb EV_0=1$, $\mathbb EV_0^2=2$ yield
$G(a)\le1-a+2a^2$.  Since $\log u\le u-1$,
$\log G(a)+a\le2a^2$.
\end{proof}

\section{Growing partitions and the uniform theorem}

\begin{definition}[Partition statistics]
Let $\lambda=1^{k_1}2^{k_2}\cdots$ be a nonempty integer partition, with
only finitely many nonzero $k_r$.  Define
\begin{equation}
 \ell=\sum_rk_r,\qquad d=\sum_rrk_r,\qquad
 R=\max\{r:k_r>0\},
 \label{eq:degree-order}
\end{equation}
and
\begin{equation}
 H=\sum_r\frac{k_r}{2r-1},\qquad
 H_2=\sum_r\frac{k_r}{(2r-1)^2}.
 \label{eq:H}
\end{equation}
Thus $0<H_2\le H\le d$ and $R\le d$.  For
$A_r\in\{P_r,J_r,I_{2r},K_r\}$ write
\[
 \cA_\lambda=\prod_r A_r^{k_r},
 \]
using the corresponding calligraphic letter.  In particular,
\begin{equation}
 \cM_\lambda(x)
 =x^{-(2d-\ell)}L^{-\ell}
  \prod_r(2r-1)^{-k_r}.
 \label{eq:Mlambda}
\end{equation}
\end{definition}

\begin{theorem}[Quantitative growing-order uniformization]
\label{thm:master}
Let $x=p_y$, $L\ge512$, and let $\lambda$ be a nonempty partition such
that $7R\eps_x\le1/2$.  Then
\begin{align}
 \left|\log\frac{\cP_\lambda}{\cJ_\lambda}\right|
 &\le14d\eps_x,
 \label{eq:ledger-J}\\
 \left|\log\frac{\cP_\lambda}{\cI_\lambda}\right|
 &\le14d\eps_x+\frac{d}{x^2-1},
 \label{eq:ledger-I}\\
 \left|\log\frac{\cP_\lambda}{\cM_\lambda}\right|
 &\le14d\eps_x+\frac{d}{x^2-1}+\frac{H}{L}.
 \label{eq:ledger-M}
\end{align}
More precisely,
\begin{equation}
 \boxed{
 \left|\log\frac{\cP_\lambda}{\cM_\lambda}+\frac{H}{L}\right|
 \le14d\eps_x+\frac{d}{x^2-1}+\frac{2H_2}{L^2}.}
 \label{eq:refined-master}
\end{equation}
\end{theorem}

\begin{proof}
Sum Proposition~\ref{prop:source-transfer} with multiplicities $k_r$;
the result is $14\eps_x\sum_r r k_r=14d\eps_x$.  Sum
Lemma~\ref{lem:J-I} to obtain $d/(x^2-1)$, and sum
\eqref{eq:I-K} to obtain $H/L$.  Finally $a_r\le1/L\le1/4$, so summing
\eqref{eq:G-refined} gives
\[
 0\le\log\frac{\cI_\lambda}{\cM_\lambda}+\frac HL
 \le\frac{2H_2}{L^2}.
\]
The triangle inequality proves \eqref{eq:refined-master}.
\end{proof}

\begin{corollary}[Asymptotic criterion]
\label{cor:criterion}
Let $\lambda_y$ be any sequence of nonempty partitions.  If
\begin{equation}
 d_y\eps_{p_y}+\frac{d_y}{p_y^2}\longrightarrow0,
 \label{eq:source-power-condition}
\end{equation}
then
$\cP_{\lambda_y}/\cJ_{\lambda_y}\to1$ and
$\cP_{\lambda_y}/\cI_{\lambda_y}\to1$.  Under
\eqref{eq:source-power-condition},
\begin{equation}
 \frac{\cP_{\lambda_y}}{\cM_{\lambda_y}}\longrightarrow1
 \quad\Longleftrightarrow\quad
 \frac{H_y}{L_y}\longrightarrow0.
 \label{eq:iff-H}
\end{equation}
Still under \eqref{eq:source-power-condition}, the explicit expansion is
\begin{equation}
 \log\frac{\cP_{\lambda_y}}{\cM_{\lambda_y}}
 =-\frac{H_y}{L_y}
 +O\!\left(d_y\eps_{p_y}+\frac{d_y}{p_y^2-1}
              +\frac{H_{2,y}}{L_y^2}\right).
 \label{eq:partition-expansion}
\end{equation}
\end{corollary}

\begin{proof}
Condition \eqref{eq:source-power-condition} implies
$7R_y\eps_{p_y}\le7d_y\eps_{p_y}\le1/2$ eventually, so
Theorem~\ref{thm:master} applies.  The first two conclusions follow from
\eqref{eq:ledger-J}--\eqref{eq:ledger-I}.  The refined estimate gives
\eqref{eq:partition-expansion}.  Since $H_2\le H$,
\[
 \frac{2H_2}{L^2}\le\frac2L\frac HL.
\]
Thus $H/L\to0$ is sufficient.  Conversely, with
$A_y=14d_y\eps_{p_y}+d_y/(p_y^2-1)$, the refined estimate gives the
explicit lower bound
\[
 \left|\log\frac{\cP_{\lambda_y}}{\cM_{\lambda_y}}\right|
 \ge\left(1-\frac2{L_y}\right)\frac{H_y}{L_y}-A_y.
\]
Here $A_y\to0$, so convergence of the ratio to one forces $H_y/L_y\to0$.
\end{proof}

\begin{corollary}[Uniform families]
\label{cor:uniform-families}
For each $y$, let $\mathfrak F_y$ be a nonempty family of finite nonempty
partitions, and suppose
\[
 D_y=\sup_{\lambda\in\mathfrak F_y}d(\lambda)<\infty,
 \qquad
 H_y^*=\sup_{\lambda\in\mathfrak F_y}H(\lambda)<\infty.
\]
If
\begin{equation}
 D_y\eps_{p_y}+\frac{D_y}{p_y^2}\longrightarrow0,
 \label{eq:uniform-source-power}
\end{equation}
then
\begin{align*}
 \sup_{\lambda\in\mathfrak F_y}
 \left|\log\frac{\cP_\lambda}{\cJ_\lambda}\right|&\longrightarrow0,\\
 \sup_{\lambda\in\mathfrak F_y}
 \left|\log\frac{\cP_\lambda}{\cI_\lambda}\right|&\longrightarrow0.
\end{align*}
Moreover,
\begin{equation}
 \sup_{\lambda\in\mathfrak F_y}
 \left|\frac{\cP_\lambda}{\cM_\lambda}-1\right|\longrightarrow0
 \quad\Longleftrightarrow\quad
 \frac{H_y^*}{L_y}\longrightarrow0.
 \label{eq:uniform-iff}
\end{equation}
\end{corollary}

\begin{proof}
Since $R(\lambda)\le d(\lambda)\le D_y$, condition
\eqref{eq:uniform-source-power} makes the source smallness hypothesis
uniformly valid.  Equations \eqref{eq:ledger-J} and \eqref{eq:ledger-I}
give the first two assertions.  Put
\[
 A_y^*=14D_y\eps_{p_y}+\frac{D_y}{p_y^2-1}.
\]
For every $\lambda\in\mathfrak F_y$, \eqref{eq:refined-master} and
$H_2(\lambda)\le H(\lambda)$ give
\[
 \left|\log\frac{\cP_\lambda}{\cM_\lambda}+\frac{H(\lambda)}{L_y}\right|
 \le A_y^*+\frac2{L_y}\frac{H(\lambda)}{L_y}.
\]
The upper bound proves sufficiency.  Taking suprema in the corresponding
reverse-triangle lower bound gives
\[
 \sup_{\lambda\in\mathfrak F_y}
 \left|\log\frac{\cP_\lambda}{\cM_\lambda}\right|
 \ge\left(1-\frac2{L_y}\right)\frac{H_y^*}{L_y}-A_y^*.
\]
Uniform convergence of the positive ratios to one is equivalent to
uniform convergence of their logarithms to zero, and necessity follows.
\end{proof}

The quantity $d\eps_x$ is a robust sufficient payment obtained from the
available source upper bound.  It is not asserted to be logically
necessary for the actual, unknown signed prime error.

\begin{corollary}[Convenient growth hypotheses]
\label{cor:sufficient}
If
\begin{equation}
 \log d_y=o(V(L_y)),\qquad H_y=o(L_y),
 \label{eq:sufficient}
\end{equation}
then $\cP_{\lambda_y}/\cM_{\lambda_y}\to1$.
\end{corollary}

\begin{proof}
Because $\log L=o(V(L))$, equation \eqref{eq:epsilon} and
$\log d=o(V)$ give $d\eps_x\to0$.  Also $V(L)=o(L)$, so
$d/x^2\to0$.  Apply Corollary~\ref{cor:criterion} and $H=o(L)$.
\end{proof}

\begin{corollary}[Single-order uniform range]
\label{cor:single-r}
Let $R_y\ge1$ be integers.  If $\log R_y=o(V(L_y))$, then
\begin{equation}
 \max_{1\le r\le R_y}
 \left|\log\frac{P_r(y)}{K_r(p_y)}\right|\longrightarrow0.
 \label{eq:single-uniform}
\end{equation}
More sharply, \eqref{eq:single-uniform} holds whenever, for some fixed
$0<\delta<0.1853$,
\begin{equation}
 R_y\le\exp\bigl((0.1853-\delta)V(L_y)\bigr)
 \label{eq:delta-range}
\end{equation}
for all sufficiently large $y$.
\end{corollary}

\begin{proof}
For $r\le R_y$, the three one-factor errors are bounded by
\[
 14R_y\eps_x+\frac{R_y}{x^2-1}+\frac1L.
\]
The first growth hypothesis makes the first two terms vanish.  Under
\eqref{eq:delta-range},
\[
 R_y\eps_x\le0.027L^{1.801}e^{-\delta V(L)}\longrightarrow0,
\]
and $R_y/x^2\to0$ because $V(L)=o(L)$.  The source smallness condition is
therefore automatic eventually.
\end{proof}

\section{Sharpness of the leading-kernel condition}

\begin{proposition}[The all-ones obstruction]
\label{prop:sharpness}
Fix $c>0$ and let
\[
 \lambda_y=1^{k_y},\qquad k_y=\lfloor cL_y\rfloor.
\]
Then
\begin{equation}
 \frac{\cP_{\lambda_y}}{\cM_{\lambda_y}}\longrightarrow e^{-c}.
 \label{eq:sharpness}
\end{equation}
\end{proposition}

\begin{proof}
Here $d_y=H_y=H_{2,y}=k_y$.  The source and power errors satisfy
$k_y\eps_{p_y}+k_y/p_y^2\to0$, while $k_y/L_y\to c$ and
$k_y/L_y^2\to0$.  Equation \eqref{eq:refined-master} therefore gives
$\log(\cP_{\lambda_y}/\cM_{\lambda_y})\to-c$.
\end{proof}

Thus a degree condition alone cannot guarantee the leading equivalent.
The statistic $H$ is not an artifact of the proof: even the simplest
growing partition detects the accumulated first correction of the
logarithmic integral kernel.
