\section{The established ledger}
\label{sec:established}

This section collects only facts that the MVP architecture imports as
proved.  Their conjunction still falls short of an \(\Ltwo\) estimate.

\begin{proposition}[\Etag\ E1: exact target anchoring]
The target defect is anchored by
\eqref{eq:hard-packet-anchor}.  Thus a proof of
\(B_{h_0,\delta}(X)=o(X)\), with the actual fixed-\(h_0\)
coefficient, is sufficient and necessary within the imported
local-model reduction.
\end{proposition}

\begin{proposition}[\Etag\ E2: determinant and full-block interfaces]
The M\"obius tail admits exact \(TT^*\), lcm-return, content,
determinant, and primitive/nonprimitive decompositions.  A generic
primitive determinant-dispersion input together with an all-block
cover would conditionally close the tail.  Neither that dispersion
input nor the all-block cover is asserted as proved
\cite{WangTPC18,WangTPC91}.
\end{proposition}

\begin{proposition}[\Etag\ E3: literal carrier typing]
On the sectors for which a literal dictionary has been certified, the
physical coefficient can be transported while retaining \(h_0\),
source keys, content/projector children, both polarizations, masks,
prefixes, M\"obius signs, actual support, and one global
normalization.  TPC-93 gives a lossless low-window affine export and
restores the reduced determinant to the physical value \(h_0\)
\cite{WangTPC93}.
\end{proposition}

\begin{proposition}[\Etag\ E4: route exclusions]
The following implications are false without additional literal
input:
\begin{enumerate}[label=(\roman*),leftmargin=2em]
\item ordinary Haar spectral behavior implies rare prime-pair
  correlation \cite{WangTPC1};
\item support, completion, or unsigned energy implies zero-mode
  cancellation \cite{WangTPC81,WangTPC82};
\item arbitrary continuous signed Mellin synthesis has a uniform
  finite lower frame \cite{WangTPC51};
\item anchor-free logarithmic cancellation implies translated
  ordinary cancellation \cite{WangTPC87,WangTPC96};
\item squarefree local density implies M\"obius-sign cancellation
  \cite{WangTPC90}; and
\item conductor \(q_X\), singleton fine cells, or bounded overlap by
  itself implies physical resonance dispersion or full-block cover
  \cite{WangTPC91,WangTPC99,WangTPC100}.
\end{enumerate}
\end{proposition}

\begin{proposition}[\Etag\ E5: two literal exceptional returns]
\label{prop:exceptional-returns}
Within the present literal low-window decomposition, the
conductor-one constant-phase branch and the nonconstant
\(q_X\mid u\) opened branch are each returned at the recorded
endpoint scale
\[
X^{o(1)}Q_X^2.
\]
These are \(\Lone\) branch closures.  They are not estimates of the
complete signed low-window coefficient
\cite{WangTPC98,WangTPC100}.
\end{proposition}

\begin{theorem}[\Etag\ E6: three-ledger resonance inequality]
\label{thm:three-ledger}
For the remaining positive invertible resonant carrier, define
\[
\PX
=\sum_{H\in\HX}\frac{2H}{q_X-1}M_H^{\mathrm{inv}},
\qquad
\XX
=\sum_{H\in\HX}g_{q_X}(H)
\sqrt{|\mathfrak X_H^\circ|},
\]
where
\[
g_q(H)
=
\sqrt{\frac{2H(q-1-2H)}{q-1}},
\]
and let \(W_X\) be the maximum normalized opened-atom weight.  The
imported literal results give
\begin{equation}
\mathcal C_{\mathrm{exc},X}^{\mathrm{abs}}
\ll
X^{o(1)}Q_X^2
+B_X\left[
\PX+
\sqrt{W_X|\HX|(q_X-1)\PX}
+\XX
\right].
\label{eq:three-ledger}
\end{equation}
The first term includes the exceptional returns in
\cref{prop:exceptional-returns}; the three quantities in brackets are
not bounded at the required growing scale by the established theory
\cite{WangTPC101,WangTPC102}.
\end{theorem}

\begin{proposition}[\Etag\ E7: two independent exponent ledgers]
\label{prop:two-ledgers}
If the determinant energy and distinguished zero mode are written as
\[
D_X\ge
X^{-\lambda_D+o(1)}Q_X^3J_X^2,
\qquad
|Z_X|\le X^{-\eta_Z+o(1)}Q_X^2,
\]
then the recorded sufficient compatibility condition is
\begin{equation}
\lambda_D\le 2\eta_Z.
\label{eq:det-compatibility}
\end{equation}
This is independent of the physical synthesis requirement
\begin{equation}
\Lambda_{\mathrm{phys}}<\frac1{400}.
\label{eq:physical-endpoint}
\end{equation}
Equality is compatible in \eqref{eq:det-compatibility} but leaves no
determinant reserve; equality in \eqref{eq:physical-endpoint} is a
stop.  The same loss may not be charged to both ledgers
\cite{WangTPC81,WangTPC82,WangTPC92}.
\end{proposition}

\begin{proposition}[\Etag\ E8: cover and arithmetic dispersion are
independent]
An exact or controlled-remainder full-block cover does not imply
primitive determinant cancellation, and a determinant estimate on a
selected block does not imply literal full-block reassembly.  Both
must be supplied in one carrier and one quantifier scope
\cite{WangTPC91,WangTPC92}.
\end{proposition}

\begin{remark}[The first genuinely new \(\Ltwo\) event]
A future theorem would count as the first \(\Ltwo\) advance only if it
reduces a fixed polynomial exponent for the actual growing
fixed-\(h_0\) coefficient, uniformly returns every sector required by
its statement, and survives the physical normalization.  A new exact
spectrum or finite certificate, however useful, remains
\(\Lzero/\Lone\).
\end{remark}
