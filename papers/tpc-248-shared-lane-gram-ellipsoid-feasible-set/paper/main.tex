\begin{abstract}
A source-index decomposition of the finite V59 twin-prime Gate-B scalar exposes,
inside each fixed output block, several probe vectors paired with one shared
physical output lane.  Independent marginal disks therefore do not describe
the source-valid joint family.  We determine that family exactly.  For the
analysis operator $V^*W=(\langle v_i,W\rangle)_i$ and Gram matrix $G=V^*V$,
the radius-$\rho$ ball maps onto
\[
 \{y\in\Ran G:y^*G^\dagger y\leq \rho^2\}.
\]
We classify the exact-sphere image by the orthogonal slack $\Ker V^*$, give the
entrywise-conjugate physical covariance form, and distinguish independent
group budgets from one global Hilbert budget.  Repeated probes collapse two
unit-disk marginals to a diagonal disk, while orthogonal probes give a Euclidean
ball rather than a bidisk.  These are exact structural results; they provide no
arithmetic cancellation, strict $1/400$ payment, full Gate B, or twin-prime
conclusion.
\end{abstract}

\section{Exact analysis-map ellipsoid}

Let $v_1,\ldots,v_m$ be ordered vectors in a finite-dimensional complex Hilbert
space $\cH$.  Define
\[
 V:\mathbb C^m\longrightarrow\cH,\qquad Va=\sum_{i=1}^m a_iv_i,
 \qquad G=V^*V,
\]
so $G_{ij}=\langle v_i,v_j\rangle$.  The Moore--Penrose inverse is denoted by
$G^\dagger$.

\begin{theorem}[ball image and minimum preimage]\label{thm:ball}
For every $\rho\geq0$,
\[
 V^*\{W\in\cH:\|W\|\leq\rho\}
 =\left\{y\in\Ran G:y^*G^\dagger y\leq\rho^2\right\}.
\]
For each $y\in\Ran G$, the vector
\[
 W_0=VG^\dagger y
\]
is the unique minimum-norm preimage and
\[
 \min_{V^*W=y}\|W\|^2=\|W_0\|^2=y^*G^\dagger y.
\]
\end{theorem}

\begin{proof}
In finite dimension, $\Ran V^*=\Ran(V^*V)=\Ran G$.  If $y\in\Ran G$, the
Moore--Penrose identities give
\[
 V^*W_0=GG^\dagger y=y.
\]
Furthermore,
\begin{align*}
 \|W_0\|^2
 &=\langle VG^\dagger y,VG^\dagger y\rangle\\
 &=(G^\dagger y)^*G(G^\dagger y)
 =y^*G^\dagger y.
\end{align*}
Here we used that $G$ and $G^\dagger$ are Hermitian and that
$GG^\dagger y=y$.  Every other preimage is uniquely $W_0+k$ with
$k\in\Ker V^*$.  Since $W_0\in\Ran V=(\Ker V^*)^\perp$,
\[
 \|W_0+k\|^2=\|W_0\|^2+\|k\|^2.
\]
The minimum formula and both inclusions in the ball identity follow.  The same
argument includes rank-zero $G$ and $\rho=0$.
\end{proof}

The range condition is essential when the probes are linearly dependent: the
quadratic inequality by itself must not be interpreted on all of
$\mathbb C^m$.

\section{Exact spheres and physical orientation}

\begin{theorem}[orthogonal-slack dichotomy]\label{thm:sphere}
Let $S_\rho=\{W:\|W\|=\rho\}$.  If $\Ker V^*\neq\{0\}$, then
\[
 V^*S_\rho=\{y\in\Ran G:y^*G^\dagger y\leq\rho^2\}.
\]
If $\Ker V^*=\{0\}$, then
\[
 V^*S_\rho=\{y\in\Ran G:y^*G^\dagger y=\rho^2\}.
\]
Both formulas remain valid for $\rho=0$.
\end{theorem}

\begin{proof}
For a point of minimum energy $e=y^*G^\dagger y\leq\rho^2$, choose a unit
$k\in\Ker V^*$ and replace $W_0$ by
$W_0+\sqrt{\rho^2-e}\,k$.  This reaches the sphere without changing $y$.
If the kernel is zero, $W_0$ is the only preimage, so sphere membership is
equivalent to $e=\rho^2$.
\end{proof}

The V59 physical scalar uses the opposite orientation
\[
 z_i=\langle W,v_i\rangle=\overline{\langle v_i,W\rangle}
 =\overline{y_i}.
\]

\begin{corollary}[physical covariance ellipsoid]\label{cor:physical}
The physical ball image is
\[
 \left\{z\in\Ran\overline G:
 z^*(\overline G)^\dagger z\leq\rho^2\right\}.
\]
The sphere classification is Theorem~\ref{thm:sphere} with $G$ replaced by
$\overline G$.
\end{corollary}

\begin{proof}
Entrywise conjugation carries $\Ran G$ onto $\Ran\overline G$, commutes with
the Moore--Penrose inverse, and conjugates the real nonnegative energy
$y^*G^\dagger y$ to the displayed expression.
\end{proof}

\section{Grouped budgets are not interchangeable}

For output groups $c$, let $V_c^*W_c=y_c$ and $G_c=V_c^*V_c$.

\begin{theorem}[product and global budgets]\label{thm:groups}
For explicitly independent constraints $\|W_c\|\leq\rho_c$, the joint image is
the Cartesian product
\[
 \prod_c\{y_c\in\Ran G_c:y_c^*G_c^\dagger y_c\leq\rho_c^2\}.
\]
For one global direct-sum constraint
$\sum_c\|W_c\|^2\leq\rho^2$, the exact image is instead
\[
 \left\{(y_c)_c: y_c\in\Ran G_c,\quad
 \sum_c y_c^*G_c^\dagger y_c\leq\rho^2\right\}.
\]
\end{theorem}

\begin{proof}
The first domain is a Cartesian product and the map acts coordinatewise.  For
the second domain, Theorem~\ref{thm:ball} supplies the minimum squared norm for
each group; direct-sum orthogonality adds these minima.  Independent minimum
preimages realize every point satisfying the summed inequality.
\end{proof}

Thus ``local ellipsoid'' does not by itself determine a cross-group feasible
set.  The budget domain is part of the theorem.

\section{Sharp adversarial fixtures}

First take $v_1=v_2=e_1$ and $\rho=1$.  Then
\[
 G=\begin{pmatrix}1&1\\1&1\end{pmatrix},\qquad
 G^\dagger=\frac14\begin{pmatrix}1&1\\1&1\end{pmatrix},
\]
and the exact image is
\[
 \{(t,t):|t|\leq1\}.
\]
Each coordinate marginal is the unit disk, but $(1,0)$ is not jointly
feasible.  This refutes the bidisk promotion at rank one.

Second take $v_1=e_1,v_2=e_2$.  The image is
$\{(y_1,y_2):|y_1|^2+|y_2|^2\leq1\}$, again strictly smaller than the bidisk.
If the ambient space contains an unused $e_3$, the exact unit sphere maps onto
that whole solid ball; without $e_3$, it maps only onto its boundary.

Finally, two scalar groups with a global unit budget have local unit-disk
marginals, but $(1,1)$ has minimum global energy $2$ and is forbidden.  These
examples isolate three different errors: ignoring Gram range, ignoring joint
energy, and ignoring orthogonal sphere slack.
