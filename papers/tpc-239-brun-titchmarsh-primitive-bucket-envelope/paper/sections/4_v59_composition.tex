\section{V59 specialization and finite-window composition}
\label{sec:composition}

\subsection{Uniform primitive-row density}

\begin{corollary}[V59 row envelope]
\label{cor:v59-row}
At the scales \eqref{eq:scales},

\begin{equation}
 \max_{\substack{h\leq U,\ a\ (\mathrm{mod}\ h)\\(a,h)=1}}
 R_h(a)
 \ll x^{1/96}\frac{\log\log x}{\log x},
 \label{eq:v59-row}
\end{equation}

where inactive rows contribute zero.
\end{corollary}

\begin{proof}
Lemma~\ref{lem:h-one} handles $h=1$.  For $2\leq h\leq U$, the exact
scale gap is

\begin{equation}
 \frac QU=x^{1/3-133/400}=x^{1/1200}.
 \label{eq:q-over-u}
\end{equation}

Consequently,

\[
 \log(2Q/h)\geq\log(2Q/U)
 =\log2+\frac{1}{1200}\log x\gg\log x.
\]

The standard maximal-order estimate
$h/\ph(h)\ll\log\log(3h)$ gives
$h/\ph(h)\ll\log\log x$ uniformly in this range.  Since

\begin{equation}
 \frac{Q^2}{H}=x^{2/3-21/32}=x^{1/96},
 \label{eq:q2h}
\end{equation}

Theorem~\ref{thm:bucket} proves \eqref{eq:v59-row}.
\end{proof}

\subsection{Substitution before the large sieve}

Let $\Ix=(x/2,x]\cap\ZZ$ and $N=\#\Ix$.  The frozen TPC-237 composition,
written with a generic maximum primitive-row multiplicity $\Rmax$, is

\begin{equation}
 \sum_{n\in\Ix}\sum_{j=1}^{J}|K_j(n)|^2
 \leq (N-1+U^2)\Rmax\cE,
 \qquad
 \cE=\sum_{h,a,j,q}|C_hB_{h,q}^{(j)}(a)|^2.
 \label{eq:frozen-composition}
\end{equation}

This inequality first compresses the $q$-rows at each primitive coordinate
and only then applies the large sieve to distinct reduced fractions $a/h$.
The order matters: passing unreduced frequencies to the large sieve would
create duplicate points.  No such change occurs here.

\begin{theorem}[Prime-density finite-window packet trace]
\label{thm:packet-trace}
For the kernel \eqref{eq:K} and fixed $J$,

\begin{equation}
 \frac1N\sum_{n\in\Ix}\sum_{j=1}^{J}|K_j(n)|^2
 \ll JM^2x^{1/48}(\log x)^4\log\log x.
 \label{eq:packet-trace}
\end{equation}

The leading unnormalized fixed-power exponent is $49/48+o(1)$.
\end{theorem}

\begin{proof}
Use Corollary~\ref{cor:v59-row} for $\Rmax$ in
\eqref{eq:frozen-composition}, and use the unchanged direct-energy estimate
\eqref{eq:direct-energy} for $\cE$.  Because $N\asymp x$,

\begin{equation}
 \frac{U^2}{N}
 =x^{2(133/400)-1+o(1)}
 =x^{-67/200+o(1)}.
 \label{eq:window-correction}
\end{equation}

After division by $N$, the two principal factors are

\[
 \left(x^{1/96}\frac{\log\log x}{\log x}\right)
 \left(JM^2x^{1/96}(\log x)^5\right).
\]

Their product is the right side of \eqref{eq:packet-trace}, since
$1/96+1/96=1/48$.  Removing the normalization multiplies by
$N=x^{1+o(1)}$, so the leading fixed-power exponent becomes
$1+1/48=49/48$.
\end{proof}

\subsection{Loss ledger}

The direct coefficient energy contributes $(\log x)^5$.  The new row
envelope contributes $\log\log x/\log x$.  Hence the final logarithmic loss
is $(\log x)^4\log\log x$.  Relative to the preceding
$(\log x)^5$ bound, the improvement factor is exactly

\[
 \frac{\log x}{\log\log x}.
\]

The fixed-power exponent remains $1/48$.  Theorem~\ref{thm:packet-trace}
therefore gives genuine logarithmic progress, not a fixed-power saving, and
does not establish sharpness of either fixed-power exponent.
