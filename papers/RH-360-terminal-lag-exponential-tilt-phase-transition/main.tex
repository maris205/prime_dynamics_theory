\begin{abstract}
We determine the full exponential-transform phase diagram of the deterministic
terminal-lag distribution isolated in RH-358.  Let
\[
 \pi_k(r)=\frac{y_k^{2k-1-r}/(2k-1-r)}{C_k},
 \qquad 0\le r\le k-2,
\]
where $y_k=x\exp[-\log C_M/k+o(k^{-1})]$ and $x>1$, and define
$G_k(z)=\sum_rz^r\pi_k(r)$.  For fixed $0\le z<x$,
\[
 G_k(z)\longrightarrow\frac{1-x^{-1}}{1-z/x}.
\]
In the critical window $z_k=x\exp(\tau/k)$,
\[
 \frac{G_k(z_k)}k\longrightarrow
 2(1-x^{-1})\int_0^1\frac{e^{(\tau+\log C_M)s}}{2-s}\,ds.
\]
For fixed $z>x$,
\[
 G_k(z)\sim
 \frac{2C_M(1-x^{-1})}{1-x/z}\left(\frac zx\right)^{k-2}.
\]
The tilted laws converge respectively to a terminal geometric distribution, a
continuous critical density on the scaled lag $r/k$, and an opposite-endpoint
geometric distribution.  Thus $k^{-1}\log G_k(z)$ has the sharp limit
$\max\{0,\log(z/x)\}$.  These are normalized deterministic budget laws, not
spectral or noisy stochastic distributions.  Actual-head inheritance remains
conditional on the original unnormalized same-clock leaf $D_{4k}(R)\to0$.
No determinant, Gate A--E, Hilbert--Polya, zero-identification, or RH claim is
made.
\end{abstract}

\section{Exact transform identity}

The source-locked radial law is
\begin{equation}\label{eq:y}
 y_k=x\exp\left[-\frac{\log C_M}{k}+o(k^{-1})\right],
 \qquad x>1,\quad C_M>0.
\end{equation}
RH-358 defines the complete deterministic strict upper budget
\begin{equation}\label{eq:C}
 C_k=\sum_{r=0}^{k-2}\frac{y_k^{2k-1-r}}{2k-1-r}
\end{equation}
and the terminal-lag probability
\begin{equation}\label{eq:pi}
 \pi_k(r)=\frac{y_k^{2k-1-r}/(2k-1-r)}{C_k},
 \qquad 0\le r\le k-2,
\end{equation}
extended by zero to $\mathbb Z_{\ge0}$
\citep{WangTerminalLag2026}.  It converges in total variation to the
geometric law $(1-x^{-1})x^{-r}$.  That convergence alone does not control
arbitrary exponential observables; the present theorem finds the exact
uniform-integrability boundary.

For $z\ge0$, define
\begin{equation}\label{eq:G}
 G_k(z):=\sum_{r=0}^{k-2}z^r\pi_k(r).
\end{equation}
For $u\ge0$, introduce the finite sum
\begin{equation}\label{eq:A}
 A_k(u):=\sum_{r=0}^{k-2}u^r\frac{2k-1}{2k-1-r}.
\end{equation}

\begin{proposition}[Exact transform quotient]\label{prop:quotient}
For every $z\ge0$,
\begin{equation}\label{eq:quotient}
 \boxed{G_k(z)=\frac{A_k(z/y_k)}{A_k(1/y_k)}.}
\end{equation}
Moreover,
\begin{equation}\label{eq:denominator}
 A_k(1/y_k)\longrightarrow\frac1{1-x^{-1}}.
\end{equation}
\end{proposition}

\begin{proof}
Multiply numerator and denominator of \eqref{eq:G} by
$(2k-1)y_k^{-(2k-1)}$ to obtain \eqref{eq:quotient}.  For each fixed $r$,
$y_k^{-r}(2k-1)/(2k-1-r)\to x^{-r}$.  Choose $1<\xi<x$ with $y_k\ge\xi$
eventually.  On the support, $(2k-1)/(2k-1-r)<2$, so the summands are bounded
by $2\xi^{-r}$.  Dominated convergence gives \eqref{eq:denominator}.
\end{proof}

\section{Subcritical, critical, and supercritical transforms}

\begin{theorem}[Three transform regimes]\label{thm:three-regimes}
The following statements hold.
\begin{enumerate}
\item If $0\le z<x$ is fixed, then
\begin{equation}\label{eq:subcritical}
 \boxed{G_k(z)\longrightarrow
 \frac{1-x^{-1}}{1-z/x}.}
\end{equation}
The convergence is locally uniform on $[0,x)$.
\item Fix $\tau\in\mathbb R$ and put
\begin{equation}\label{eq:critical-z}
 z_k=x\exp(\tau/k),
 \qquad \eta:=\tau+\log C_M.
\end{equation}
Then
\begin{equation}\label{eq:critical}
 \boxed{
 \frac{G_k(z_k)}k\longrightarrow
 \Phi(\eta):=
 2(1-x^{-1})\int_0^1\frac{e^{\eta s}}{2-s}\,ds.}
\end{equation}
\item If $z>x$ is fixed, then
\begin{equation}\label{eq:supercritical}
 \boxed{
 G_k(z)\sim
 \frac{2C_M(1-x^{-1})}{1-x/z}
 \left(\frac zx\right)^{k-2}.}
\end{equation}
\end{enumerate}
\end{theorem}

\begin{proof}
For the first statement, $z/y_k\to z/x<1$.  On any compact subinterval below
$x$, the numerator in \eqref{eq:quotient} is dominated by one common
geometric sequence.  Thus
\[
 A_k(z/y_k)\longrightarrow\frac1{1-z/x};
\]
combine with \eqref{eq:denominator}.

In the critical window, \eqref{eq:y} gives
\begin{equation}\label{eq:critical-u}
 k\log(z_k/y_k)\longrightarrow\eta.
\end{equation}
Uniformly for $0\le r\le k-2$,
\[
 (z_k/y_k)^r=e^{\eta r/k+o(1)},
 \qquad
 \frac{2k-1}{2k-1-r}
 =\frac{2}{2-r/k}+o(1).
\]
Therefore the Riemann sum satisfies
\[
 \frac1kA_k(z_k/y_k)
 \longrightarrow2\int_0^1\frac{e^{\eta s}}{2-s}\,ds.
\]
Division by \eqref{eq:denominator} proves \eqref{eq:critical}.

For $z>x$, write $u_k=z/y_k\to z/x>1$ and reverse the numerator sum:
\begin{equation}\label{eq:reverse-super}
 A_k(u_k)=u_k^{k-2}\frac{2k-1}{k+1}
 \sum_{\ell=0}^{k-2}u_k^{-\ell}\frac{k+1}{k+1+\ell}.
\end{equation}
Dominated convergence gives a sum limit $(1-x/z)^{-1}$, while the prefactor
ratio tends to two.  Finally,
\begin{equation}\label{eq:super-power}
 u_k^{k-2}\sim C_M(z/x)^{k-2}
\end{equation}
by \eqref{eq:y}.  Use \eqref{eq:denominator} to obtain
\eqref{eq:supercritical}.
\end{proof}

\begin{corollary}[Free-energy transition]\label{cor:free-energy}
For every fixed $z\ge0$,
\begin{equation}\label{eq:free-energy}
 \boxed{
 \lim_{k\to\infty}\frac1k\log G_k(z)
 =\max\left\{0,\log\frac zx\right\},}
\end{equation}
where the right side is interpreted as zero at $z=0$.
\end{corollary}

\begin{proof}
For $z<x$, \eqref{eq:subcritical} has a finite positive limit.  At $z=x$,
use \eqref{eq:critical} with $\tau=0$ to get $G_k(x)\sim k\Phi(\log C_M)$.
For $z>x$, take logarithms in \eqref{eq:supercritical}.
\end{proof}

\begin{remark}[The multiplier drift is critical]
The subcritical transform forgets $C_M$.  At $z=x$, the leading constant is
$\Phi(\log C_M)$, and above $x$ the factor $C_M$ appears explicitly.  Replacing
$y_k$ by $x$ before taking the critical or supercritical limit loses a leading
constant.
\end{remark}

\section{The three tilted distribution limits}

Whenever $G_k(z)>0$, define the mathematical tilted probability
\begin{equation}\label{eq:tilted}
 \pi_{k,z}(r):=\frac{z^r\pi_k(r)}{G_k(z)}.
\end{equation}

\begin{theorem}[Tilted phase diagram]\label{thm:tilted}
The tilted laws have the following limits.
\begin{enumerate}
\item For fixed $0\le z<x$, extended by zero off the finite support,
\begin{equation}\label{eq:sub-tilt}
 \pi_{k,z}\longrightarrow
 (1-z/x)(z/x)^r
\end{equation}
in total variation on $\mathbb Z_{\ge0}$, with the $z=0$ law interpreted as
$\delta_0$.
\item For $z_k=x\exp(\tau/k)$, the scaled lag $R_k/k$ under
$\pi_{k,z_k}$ converges weakly on $[0,1]$ to the density
\begin{equation}\label{eq:critical-density}
 f_\eta(s)=
 \frac{e^{\eta s}/(2-s)}
 {\displaystyle\int_0^1e^{\eta u}/(2-u)\,du},
 \qquad \eta=\tau+\log C_M.
\end{equation}
\item For fixed $z>x$, let $L_k=k-2-R_k$.  Then
\begin{equation}\label{eq:super-tilt}
 \mathbb P_{\pi_{k,z}}(L_k=\ell)
 \longrightarrow(1-x/z)(x/z)^\ell
\end{equation}
in total variation on $\mathbb Z_{\ge0}$.
\end{enumerate}
\end{theorem}

\begin{proof}
The tilted unnormalized weights are
\begin{equation}\label{eq:tilted-weights}
 (z/y_k)^r\frac{2k-1}{2k-1-r}.
\end{equation}
Below $x$, these weights converge pointwise to $(z/x)^r$ and have a common
summable geometric envelope.  Dominated convergence before and after
normalization proves \eqref{eq:sub-tilt}.

In the critical window, for every continuous $g$ on $[0,1]$, the same Riemann
sum used in \eqref{eq:critical} gives
\[
 \frac1k\sum_{r=0}^{k-2}g(r/k)
 (z_k/y_k)^r\frac{2k-1}{2k-1-r}
 \longrightarrow2\int_0^1g(s)\frac{e^{\eta s}}{2-s}\,ds.
\]
Divide by the case $g=1$ to obtain \eqref{eq:critical-density}.

Above $x$, substitute $r=k-2-\ell$ into \eqref{eq:tilted-weights} and divide by
the $\ell=0$ weight.  The ratio tends to $(x/z)^\ell$ and is dominated by a
common geometric sequence.  Normalization proves \eqref{eq:super-tilt}.
\end{proof}

\begin{remark}[No spectral probability claim]
The probabilities in \eqref{eq:tilted} are normalized deterministic absolute
budget weights.  They are not eigenvalue distributions, root-counting
measures, Gibbs states of the noisy operator, or local probability laws for
the dynamics.
\end{remark}

\section{Conditional actual-head inheritance}

Let $\pi_k^{\cH}$ be the actual even-weight lag distribution defined in
RH-358.  Under the original same-clock unnormalized leaf
\begin{equation}\label{eq:D}
 D_{4k}(R):=\sum_{2\le n<4k}
 \frac{|h_{\sigma,n}-s_{k,n}|R^n}{n}\longrightarrow0,
\end{equation}
RH-358 proves the stronger coordinatewise conclusion
\begin{equation}\label{eq:coordinate-transfer}
 \delta_k:=\sup_{0\le r\le k-2}
 \left|\frac{\pi_k^{\cH}(r)}{\pi_k(r)}-1\right|\longrightarrow0.
\end{equation}
For any sequence $z_k\ge0$, put
\begin{equation}\label{eq:actual-G}
 G_k^{\cH}(z_k):=\sum_rz_k^r\pi_k^{\cH}(r).
\end{equation}

\begin{theorem}[Uniform conditional tilt transfer]\label{thm:conditional}
Assume \eqref{eq:D}.  For every nonnegative tilt sequence,
\begin{equation}\label{eq:G-transfer}
 \left|\frac{G_k^{\cH}(z_k)}{G_k(z_k)}-1\right|\le\delta_k\longrightarrow0.
\end{equation}
Moreover, if $z_k>0$ and $\pi_{k,z_k}^{\cH}$ is the normalized actual tilted
law, then
\begin{equation}\label{eq:tilt-transfer}
 \sup_r\left|
 \frac{\pi_{k,z_k}^{\cH}(r)}{\pi_{k,z_k}(r)}-1
 \right|\longrightarrow0.
\end{equation}
If $z_k=0$, both tilted laws are exactly $\delta_0$ whenever they are defined.
Consequently every transform and tilted-law conclusion above transfers only
under \eqref{eq:D}.
\end{theorem}

\begin{proof}
The ratio in \eqref{eq:G-transfer} is a weighted average of
$\pi_k^{\cH}(r)/\pi_k(r)$ with nonnegative weights $z_k^r\pi_k(r)$, so its
distance from one is at most $\delta_k$.  Equation \eqref{eq:tilt-transfer}
follows by dividing the coordinatewise ratio by
$G_k^{\cH}(z_k)/G_k(z_k)$.
\end{proof}

The hypothesis \eqref{eq:D} remains open.  This conditional theorem transfers
budget laws only and contains no root, rank, or determinant identification.
