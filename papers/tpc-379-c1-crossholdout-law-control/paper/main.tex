\section{Finite operator and four laws}

For $I=[a,a+N-1]\cap\mathbb Z$, $p\in(Q,2Q]$, and $u,t\in I$, put
\[
 K_p(u,t)=p\left(\frac pQ\right)^2
 \frac{66^2}{66^2+(u-t)^2}
 \left({\bf1}_{p\mid(u-t)}-\frac1{p-1}\right)
 \mathbf{1}_{u\ne t}\mathbf{1}_{p\nmid u}\mathbf{1}_{p\nmid t}.
\]
For a law $\ell$ with signs $s_p(\ell)\in\{\!-1,1\}$, define
\[
 A_\ell(u,t)=\sum_{Q<p\leq2Q}s_p(\ell)K_p(u,t),\qquad
 G(u)=\sum_{t\in I}\sum_{Q<p\leq2Q}K_p(u,t)^2,
\]
\[
 T_\ell(u,t)=\frac{A_\ell(u,t)}{\sqrt{G(u)G(t)}}.
\]
The geometry $G$ is common to all four laws. The signs are fixed in the
ordered prime shell as follows:
\[
\begin{array}{c|l}
\ell&s_p(\ell)\\ \hline
\text{all\_plus}&1\\
\text{alternating\_index}&(-1)^{\operatorname{index}(p)}\\
\text{mod4\_character}&1\text{ if }p\equiv1\pmod4,\ -1\text{ otherwise}\\
\text{half\_split}&1\text{ on the first shell half, }-1\text{ on the second}
\end{array}
\]
For $b(u)=\lfloor(u-a)/256\rfloor$, use the common band and tail
\[
 B_\ell(u,t)=T_\ell(u,t)\mathbf{1}_{|b(u)-b(t)|\leq1},\qquad
 R_\ell=T_\ell-B_\ell.
\]
For any unit eigenvector $v$ of $T_\ell$ with eigenvalue $\lambda$,
\[
 v^{\mathsf T}B_\ell v+v^{\mathsf T}R_\ell v=\lambda
\]
is an exact finite identity.
