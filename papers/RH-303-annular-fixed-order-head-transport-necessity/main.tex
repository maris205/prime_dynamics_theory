\begin{abstract}
An aggregate annular theorem would bypass root-by-root matching, but it would
not bypass the underlying fixed-order moment transport.  We make this
necessity quantitative.  A vanishing $H^\infty$ or $H^2$ norm of the actual
complement-to-anchor logarithmic mismatch forces every fixed coefficient to
converge.  Combined with the archived fixed-order total-trace and
counterloop limits, this implies convergence of every fixed noisy-head
moment to the finite-radius counterloop moment.  The result is a necessity
theorem, not an actual moving-head estimate.
\end{abstract}

\section{Typed decomposition}
Let $c_{\sigma,n}$ be the total noisy bulk trace, $h_{\sigma,n}$ the moment
of the modulus-complete noisy head, and
\[
 \tau_{\sigma,n}=c_{\sigma,n}-h_{\sigma,n}.
\]
Let $c_n$ be the deterministic bulk trace, $p_n$ the limiting pole moment,
$a_n=c_n-p_n$, and $s_{k,n}$ the finite-radius counterloop moment.  The
archived fixed-order inputs are
\[
 c_{\sigma,n}\to c_n,\qquad s_{k_\sigma,n}\to p_n
 \quad(k_\sigma\to\infty)
\]
for each fixed $n$.  Define
\[
 g_\sigma(z)=\sum_{n\ge2}
 \frac{\tau_{\sigma,n}-a_n}{n}z^n.
\]

\section{Coefficient and transport theorem}
\begin{theorem}\label{thm:necessity}
Fix $\rho>0$ on which $g_\sigma$ belongs to
$X=H^\infty(\rho)$ or $H^2(\rho)$.  For every fixed $n\ge2$,
\[
 |\tau_{\sigma,n}-a_n|
 \le n\rho^{-n}\|g_\sigma\|_X.
\]
Moreover,
\begin{align*}
 |h_{\sigma,n}-s_{k_\sigma,n}|
 \le{}& |c_{\sigma,n}-c_n|
 +|s_{k_\sigma,n}-p_n|\\
 &+n\rho^{-n}\|g_\sigma\|_X.
\end{align*}
Consequently, annular convergence together with the two archived fixed-order
limits forces
\[
 h_{\sigma,n}-s_{k_\sigma,n}\longrightarrow0
\]
for every fixed order.
\end{theorem}
\begin{proof}
The coefficient of $z^n$ in $g_\sigma$ is
$(\tau_{\sigma,n}-a_n)/n$.  Cauchy's estimate proves the $H^\infty$ case;
in $H^2$, any one boundary-scaled coefficient is bounded by the square norm.
For the second display, rearrange the exact identity
\[
 \tau_{\sigma,n}-a_n
 =(c_{\sigma,n}-c_n)-(h_{\sigma,n}-p_n)
\]
as
\[
 h_{\sigma,n}-s_{k_\sigma,n}
 =(c_{\sigma,n}-c_n)-(\tau_{\sigma,n}-a_n)
 -(s_{k_\sigma,n}-p_n)
\]
and apply the triangle inequality.
\end{proof}

\section{Protocol and boundary}
The computation records the coefficient factor $n\rho^{-n}$ at
$\rho=1.41$ for orders $2,3,5,8$ and checks the three-term transport bound.
These are scale diagnostics only; no noisy head moments are fitted.

\begin{remark}
Aggregate annular convergence can eliminate the need to pair individual
roots, but it still forces aggregate head moments to follow the counterloop
at every fixed order.  The theorem does not prove such transport on a
growing clock.  Gates A--E remain false/open and no arithmetic divisor or RH
claim follows.
\end{remark}
